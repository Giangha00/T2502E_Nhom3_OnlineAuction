===== 4.2.4 Place Bid Flow (Luồng Đặt giá/Trả giá) =====
@startuml
title 4. Place Bid Flow — OnlineAuction (/Auction/PlaceBid)

skinparam ConditionEndStyle hline

|Người mua|
start
:Xem /Auction/Detail/{id}\n(phiên live / ending_soon);
:Nhập số tiền muốn đặt;
:Nhấn "Place Bid"\n(POST /Auction/PlaceBid);

|Hệ thống|

  :401 — Please sign in\nto place a bid;
else (Có)
  if (User bị ShadowBan?) then (Có)
    :403 + notification BidFailed;
  else (Không)
    :Check Rate Limit\n(user / auction / IP);

    if (Vượt rate limit?) then (Có)
      :429 + notification BidFailed;
    else (Không)
      if (Yêu cầu Challenge\nvà token không hợp lệ?) then (Có)
        :403 — requiresChallenge;
      else (Không / token OK)
        :PlaceBidAsync\n(transaction Serializable);
        :ValidateBid:\n- Không phải seller\n- Status = live/ending_soon\n- Trong Start–End\n- amount ≥ Current + BidStep\n- tăng đúng bội BidStep;

        if (Validate OK?) then (Không)
          :400 — lỗi (ended /\nconfirming / giá thấp...);
        else (Có)
          if (Cần đăng ký/đặt cọc\nmà chưa đủ điều kiện?) then (Có)
            :Chặn bid\n(GetBidBlockMessage);
          else (Không)
            :Fraud pre-bid gate;

            if (Fraud cho phép?) then (Không)
              :403 — fraud protection;
            else (Có)
              :Lưu Bid mới (IsWinning);
              :Bỏ IsWinning bid cũ;
              :Cập nhật CurrentPrice;
              if (Anti-snipe\nđược áp dụng?) then (Có)
                :Gia hạn EndDate;
              else (Không)
              endif
              :Commit DB;
              :RabbitMQ BidsPlaced\n→ outbid notification;
              :SignalR BidUpdated\n(giá, history, minNextBid);
              :JSON success\n"Bid placed successfully.";
            endif
          endif
        endif
      endif
    endif
  endif
endif

stop
@enduml

===== 4.2.5 Auction Closing Flow (Luồng Kết thúc Đấu giá) =====
@startuml
title 5. Auction Closing Flow — OnlineAuction

skinparam ConditionEndStyle hline

|Background Worker|
start
:AuctionFinalizationWorker\npoll mỗi ~15 giây;
:PublishTick → RabbitMQ\nAuctionLifecycle\n(FinalizeExpiredAuctions);

|Hệ thống|
:OrderCreationService\n.FinalizeExpiredAuctionsAsync;
:Tìm auction hết giờ:\nListingType=auction\nStatus ∈ {live, ending_soon}\nEndDate ≤ UtcNow;

if (Có auction cần finalize?) then (Không)
  :Kết thúc tick này;
else (Có)
  repeat
    :CreatePendingPaymentOrder\nForAuctionAsync\n(transaction Serializable);
    :Re-check status + EndDate;

    if (Vẫn cần finalize?) then (Không)
      :Skip (đã xử lý /\nchưa hết giờ);
    else (Có)
      :Lấy Bid IsWinning\n(cao nhất);

      if (Có winning bid?) then (Không)
        :Status = ended\n(WinnerId = null);
        :Commit;
      else (Có)
        :Tạo Order + OrderItem\norder_source = auction_win\nstatus = pending_payment\ndeadline = 48 giờ;
        :Status = awaiting_payment;
        :WinnerId = bidder thắng;
        :Commit;
        if (Cọc winner > giá thắng?) then (Có)
          :Refund phần cọc dư\n(PayPal);
        else (Không)
        endif
        :NotifyAuctionWin\n(winner — in-app + FCM)\n+ link /Order;
      endif
    endif

    :RefundLoserDeposits\nForAuctionAsync\n(hoàn cọc người thua,\nbỏ qua winner);
  repeat while (Còn auction\ntrong batch?) is (Có)
  ->Không;
endif

stop
@enduml
